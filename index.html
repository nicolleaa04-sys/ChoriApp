function initGeolocation() { const title = document.getElementById('geoStatusTitle'); const desc = document.getElementById('geoStatusDesc'); const warning = document.getElementById('gpsWarning'); title.textContent = "Buscando tu ubicación..."; desc.textContent = "Obteniendo coordenadas GPS..."; warning.style.display = 'none'; if (!navigator.geolocation) { title.textContent = "GPS no compatible"; desc.textContent = "Tu navegador no soporta geolocalización"; warning.style.display = 'block'; return; } navigator.geolocation.getCurrentPosition( (pos) => { const lat = pos.coords.latitude.toFixed(4); const lng = pos.coords.longitude.toFixed(4); document.getElementById('coordDisplay').textContent = `${lat}, ${lng}`; title.textContent = "¡Ubicación detectada!"; desc.textContent = `Precisión aprox.: ±${Math.round(pos.coords.accuracy)} m`; warning.style.display = 'none'; }, (err) => { console.warn("Error GPS:", err); let errorMsg = "Dentro del centro comercial puedes escribir una referencia manual."; if (err.code === 1) { // PERMISSION_DENIED title.textContent = "Permiso de ubicación denegado"; errorMsg = "Por favor, activa los permisos de ubicación para este sitio en tu navegador."; } else if (err.code === 2) { // POSITION_UNAVAILABLE title.textContent = "Señal GPS no disponible"; } else if (err.code === 3) { // TIMEOUT title.textContent = "Tiempo de espera agotado"; } desc.textContent = "Usando referencia aproximada"; warning.innerHTML = `⚠️ Señal no obtenida${errorMsg}`; warning.style.display = 'block'; }, { enableHighAccuracy: true, // Fuerza a usar GPS real y Wi-Fi de alta precisión timeout: 15000, // Espera hasta 15 segundos para dar tiempo a conectar maximumAge: 10000 // Acepta una ubicación reciente de hasta 10 segundos } ); }
